@dp.message(Command("start"))
async def send_welcome(message: Message) -> None:
    answer: str = ("Привет!"
                   "\nЯ - бот для просмотра кино 24frames."
                   "\nНапиши название фильма или сериала, который ты хочешь посмотреть.")
    await message.answer(answer)


@dp.message(Command("help"))
async def get_help(message: Message) -> None:
    answer: str = ("Введи название для просмотра."
                   "\n/help - помощь"
                   "\n/history - история поиска"
                   "\n/stats - сколько раз предлагался тот или иной фильм")
    await message.answer(answer)


@dp.message(Command("history"))
async def get_history(message: Message) -> None:
    try:
        if message.from_user:
            history: list[str] = await collector.get_history(message.from_user.id)
            answer: str = f"Ваша история запросов:{os.linesep}{os.linesep}{os.linesep.join(history)}"\
                if history else "Ваша история запросов пуста"
            await send_long_message(message, answer)
    except ConnectionError:
        await message.answer("История поиска временно недоступна.")


@dp.message(Command("stats"))
async def get_stats(message: Message) -> None:
    try:
        if message.from_user:
            stats: list[tuple[str, int]] = await collector.get_stats(message.from_user.id)
            strs: list[str] = [str(amount) + '\t' + name for name, amount in stats]
            answer: str = f"Ваша статистика запросов:{os.linesep}{os.linesep}{os.linesep.join(strs)}"\
                if strs else "Ваша статистика запросов пуста"
            await send_long_message(message, answer)
    except ConnectionError:
        await message.answer("Статистика временно недоступна.")


@dp.message()
async def get_film(message: Message) -> None:
    try:
        film: Film = await collector.get_film_with_collection(message.chat.id, message.text if message.text else "")
        answer: str = film.to_string()
        try:
            if message.bot:
                await message.bot.send_photo(chat_id=message.chat.id,
                                             photo=film.pic,
                                             caption=answer,
                                             parse_mode='Markdown')
        except TelegramBadRequest:
            if message.bot:
                await message.bot.send_photo(chat_id=message.chat.id,
                                             photo='https://cdn.culture.ru/images/a629a2fe-dc84-5cc1-913e-ee2c3377b5fa',
                                             caption=answer,
                                             parse_mode='Markdown')
    except (FilmNotFoundError, OriginError) as err:
        await message.answer(str(err))
